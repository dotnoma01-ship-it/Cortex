import {
  createContext,
  useCallback,
  useContext,
  useEffect,
  useMemo,
  useState,
  type ReactNode,
} from "react";
import { getProduct, type Product } from "@/data/catalog";

export type CartLine = { handle: string; variantId: number; qty: number };

type Persisted = {
  cart: CartLine[];
  wishlist: string[];
  compare: string[];
  currency: string;
  code: string | null;
  user: { name: string; email: string } | null;
};

type Store = Persisted & {
  cartOpen: boolean;
  searchOpen: boolean;
  quick: string | null;
  menuOpen: boolean;
  toast: string | null;
  offersOpen: boolean;
  setCartOpen: (v: boolean) => void;
  setSearchOpen: (v: boolean) => void;
  setQuick: (h: string | null) => void;
  setMenuOpen: (v: boolean) => void;
  setOffersOpen: (v: boolean) => void;
  setCurrency: (c: string) => void;
  notify: (m: string) => void;
  add: (handle: string, variantId: number, qty?: number) => void;
  setQty: (variantId: number, qty: number) => void;
  remove: (variantId: number) => void;
  clearCart: () => void;
  toggleWish: (handle: string) => void;
  wished: (handle: string) => boolean;
  toggleCompare: (handle: string) => void;
  clearCompare: () => void;
  applyCode: (code: string) => boolean;
  clearCode: () => void;
  signIn: (name: string, email: string) => void;
  signOut: () => void;
  count: number;
  subtotal: number;
  discount: number;
  total: number;
};

const KEY = "garace-hyper-v1";
const Ctx = createContext<Store | null>(null);

function load(): Persisted {
  try {
    const raw = localStorage.getItem(KEY);
    if (raw) {
      const p = JSON.parse(raw) as Persisted;
      return {
        cart: p.cart ?? [],
        wishlist: p.wishlist ?? [],
        compare: p.compare ?? [],
        currency: p.currency ?? "USD",
        code: p.code ?? null,
        user: p.user ?? null,
      };
    }
  } catch {
    /* ignore */
  }
  return { cart: [], wishlist: [], compare: [], currency: "USD", code: null, user: null };
}

export function linePrice(line: CartLine) {
  const p = getProduct(line.handle);
  const v = p?.variants.find((x) => x.id === line.variantId) ?? p?.variants[0];
  return v ? v.price * line.qty : 0;
}

export function volumeRate(qty: number) {
  if (qty >= 10) return 0.1;
  if (qty >= 5) return 0.05;
  return 0;
}

export function StoreProvider({ children }: { children: ReactNode }) {
  const [persisted, setPersisted] = useState<Persisted>(load);
  const [cartOpen, setCartOpen] = useState(false);
  const [searchOpen, setSearchOpen] = useState(false);
  const [quick, setQuick] = useState<string | null>(null);
  const [menuOpen, setMenuOpen] = useState(false);
  const [offersOpen, setOffersOpen] = useState(false);
  const [toast, setToast] = useState<string | null>(null);

  useEffect(() => {
    localStorage.setItem(KEY, JSON.stringify(persisted));
  }, [persisted]);

  const notify = useCallback((m: string) => {
    setToast(m);
    window.setTimeout(() => setToast(null), 2400);
  }, []);

  const add = useCallback(
    (handle: string, variantId: number, qty = 1) => {
      const p = getProduct(handle);
      const v = p?.variants.find((x) => x.id === variantId);
      if (!p || !v || !v.available || p.tags.includes("Coming Soon") || p.stock === 0) {
        notify("This piece is not available to add right now.");
        return;
      }
      setPersisted((s) => {
        const i = s.cart.findIndex((l) => l.variantId === variantId);
        const cart = [...s.cart];
        if (i >= 0) cart[i] = { ...cart[i], qty: cart[i].qty + qty };
        else cart.push({ handle, variantId, qty });
        return { ...s, cart };
      });
      setCartOpen(true);
      notify(`${p.title} added to cart`);
    },
    [notify],
  );

  const setQty = useCallback((variantId: number, qty: number) => {
    setPersisted((s) => ({
      ...s,
      cart:
        qty <= 0
          ? s.cart.filter((l) => l.variantId !== variantId)
          : s.cart.map((l) => (l.variantId === variantId ? { ...l, qty } : l)),
    }));
  }, []);

  const remove = useCallback((variantId: number) => setQty(variantId, 0), [setQty]);
  const clearCart = useCallback(() => setPersisted((s) => ({ ...s, cart: [], code: null })), []);

  const toggleWish = useCallback(
    (handle: string) => {
      setPersisted((s) => {
        const has = s.wishlist.includes(handle);
        return {
          ...s,
          wishlist: has ? s.wishlist.filter((h) => h !== handle) : [...s.wishlist, handle],
        };
      });
      const p = getProduct(handle);
      notify(
        persisted.wishlist.includes(handle)
          ? "Removed from wishlist"
          : `${p?.title ?? "Piece"} saved`,
      );
    },
    [notify, persisted.wishlist],
  );

  const wished = useCallback(
    (handle: string) => persisted.wishlist.includes(handle),
    [persisted.wishlist],
  );

  const toggleCompare = useCallback(
    (handle: string) => {
      setPersisted((s) => {
        if (s.compare.includes(handle)) return { ...s, compare: s.compare.filter((h) => h !== handle) };
        if (s.compare.length >= 5) return s;
        return { ...s, compare: [...s.compare, handle] };
      });
    },
    [],
  );

  const clearCompare = useCallback(() => setPersisted((s) => ({ ...s, compare: [] })), []);

  const applyCode = useCallback(
    (code: string) => {
      const c = code.trim().toUpperCase();
      if (c === "FLASH30" || c === "WELCOME10") {
        setPersisted((s) => ({ ...s, code: c }));
        notify(`Code ${c} applied`);
        return true;
      }
      notify("That code is not recognized");
      return false;
    },
    [notify],
  );

  const clearCode = useCallback(() => setPersisted((s) => ({ ...s, code: null })), []);
  const setCurrency = useCallback(
    (currency: string) => setPersisted((s) => ({ ...s, currency })),
    [],
  );
  const signIn = useCallback((name: string, email: string) => {
    setPersisted((s) => ({ ...s, user: { name, email } }));
    notify(`Welcome, ${name.split(" ")[0]}`);
  }, [notify]);
  const signOut = useCallback(() => setPersisted((s) => ({ ...s, user: null })), []);

  const subtotal = persisted.cart.reduce((n, l) => n + linePrice(l), 0);
  const volumeOff = persisted.cart.reduce((n, l) => n + linePrice(l) * volumeRate(l.qty), 0);
  const codeRate = persisted.code === "FLASH30" ? 0.3 : persisted.code === "WELCOME10" ? 0.1 : 0;
  const discount = volumeOff + (subtotal - volumeOff) * codeRate;
  const total = Math.max(0, subtotal - discount);
  const count = persisted.cart.reduce((n, l) => n + l.qty, 0);

  const value = useMemo<Store>(
    () => ({
      ...persisted,
      cartOpen,
      searchOpen,
      quick,
      menuOpen,
      toast,
      offersOpen,
      setCartOpen,
      setSearchOpen,
      setQuick,
      setMenuOpen,
      setOffersOpen,
      setCurrency,
      notify,
      add,
      setQty,
      remove,
      clearCart,
      toggleWish,
      wished,
      toggleCompare,
      clearCompare,
      applyCode,
      clearCode,
      signIn,
      signOut,
      count,
      subtotal,
      discount,
      total,
    }),
    [
      persisted,
      cartOpen,
      searchOpen,
      quick,
      menuOpen,
      toast,
      offersOpen,
      setCurrency,
      notify,
      add,
      setQty,
      remove,
      clearCart,
      toggleWish,
      wished,
      toggleCompare,
      clearCompare,
      applyCode,
      clearCode,
      signIn,
      signOut,
      count,
      subtotal,
      discount,
      total,
    ],
  );

  return <Ctx.Provider value={value}>{children}</Ctx.Provider>;
}

export function useStore() {
  const v = useContext(Ctx);
  if (!v) throw new Error("store");
  return v;
}

export function useMoney() {
  const { currency } = useStore();
  return currency;
}

function QuickView() {
  const s = useStore();
  const p = s.quick ? getProduct(s.quick) : undefined;
  const colors = p ? colorOption(p) : undefined;
  const [color, setColor] = useState<string | undefined>();
  const [mat, setMat] = useState<string | undefined>();
  const [qty, setQty] = useState(1);
  useEffect(() => {
    if (!p) return;
    setColor(colorOption(p)?.values[0]);
    const other = p.options.find((o) => o.name !== colorOption(p)?.name);
    setMat(other?.values[0]);
    setQty(1);
  }, [p]);
  if (!s.quick || !p) return null;
  const other = p.options.find((o) => o.name !== colors?.name);
  const variant = pickVariant(p, {
    ...(colors && color ? { [colors.name]: color } : {}),
    ...(other && mat ? { [other.name]: mat } : {}),
  });
  const coming = p.tags.includes("Coming Soon") || p.stock === 0;
  return (
    <div className="fixed inset-0 z-[60] grid place-items-center p-4">
      <button className="absolute inset-0 bg-black/45" aria-label="Close quick view" onClick={() => s.setQuick(null)} />
      <div className="rise relative grid max-h-[90vh] w-full max-w-4xl overflow-auto rounded-3xl bg-white md:grid-cols-2">
        <button type="button" className="absolute right-4 top-4 z-10" aria-label="Close" onClick={() => s.setQuick(null)}>
          <CloseIcon />
        </button>
        <img src={img(variant.image || p.images[0], 1000)} alt={p.title} className="h-full max-h-[70vh] w-full object-cover bg-[#f3f3f3]" referrerPolicy="no-referrer" />
        <div className="p-6 md:p-8">
          <p className="text-xs font-bold uppercase tracking-wide text-[#888]">{p.type}</p>
          <h3 className="mt-1 text-2xl font-bold">{p.title}</h3>
          <div className="mt-2">
            <Money amount={variant.price} compare={variant.compare} />
          </div>
          <p className="mt-3 text-sm text-[#555]">{p.desc}</p>
          {colors ? (
            <div className="mt-4">
              <p className="mb-2 text-sm font-bold">Color: {color}</p>
              <div className="flex flex-wrap gap-2">
                {colors.values.map((c) => (
                  <Swatch key={c} name={c} active={color === c} onClick={() => setColor(c)} image={findVariant(p, c).image} />
                ))}
              </div>
            </div>
          ) : null}
          {other ? (
            <div className="mt-4">
              <p className="mb-2 text-sm font-bold">{other.name}: {mat}</p>
              <div className="flex flex-wrap gap-2">
                {other.values.map((m) => (
                  <button key={m} type="button" onClick={() => setMat(m)} className={`rounded-full border px-3 py-1.5 text-sm font-semibold ${mat === m ? "border-black bg-black text-white" : "border-[#e5e5e5]"}`}>
                    {m}
                  </button>
                ))}
              </div>
            </div>
          ) : null}
          <div className="mt-5 flex flex-wrap items-center gap-3">
            <Qty value={qty} onChange={setQty} label={p.title} />
            <Btn
              disabled={coming || !variant.available}
              onClick={() => {
                s.add(p.handle, variant.id, qty);
                s.setQuick(null);
              }}
            >
              {coming ? "Coming soon" : variant.available ? "Add to cart" : "Sold out"}
            </Btn>
          </div>
          <Link to={`/products/${p.handle}`} className="mt-4 inline-block text-sm font-bold underline" onClick={() => s.setQuick(null)}>
            View full details
          </Link>
        </div>
      </div>
    </div>
  );
}

function MobileSearch() {
  const s = useStore();
  const nav = useNavigate();
  const [q, setQ] = useState("");
  const hits = useMemo(
    () =>
      products
        .filter((p) => `${p.title} ${p.type}`.toLowerCase().includes(q.toLowerCase()))
        .slice(0, 8),
    [q],
  );
  if (!s.searchOpen) return null;
  return (
    <div className="fixed inset-0 z-[60] bg-white p-4">
      <form
        className="flex items-center gap-2"
        onSubmit={(e) => {
          e.preventDefault();
          s.setSearchOpen(false);
          nav(`/search?q=${encodeURIComponent(q)}`);
        }}
      >
        <input autoFocus value={q} onChange={(e) => setQ(e.target.value)} placeholder="Search" className="h-12 flex-1 rounded-full bg-[#ededed] px-5 outline-none" />
        <button type="button" aria-label="Close search" onClick={() => s.setSearchOpen(false)}>
          <CloseIcon />
        </button>

        {hits.map((p) => (
          <Link key={p.handle} to={`/products/${p.handle}`} onClick={() => s.setSearchOpen(false)} className="flex items-center gap-3">

            <span className="font-semibold">{p.title}</span>
          </Link>
        ))}
      </div>
    </div>
  );
}

function Offers() {
  const s = useStore();
  const [copied, setCopied] = useState("");

    navigator.clipboard?.writeText(code).catch(() => {});
    setCopied(code);
    s.applyCode(code);
  };
  return (
    <div className="fixed bottom-24 left-4 z-30 hidden md:block">
      {s.offersOpen ? (
        <div className="mb-2 w-72 rounded-3xl bg-white p-4 shadow-2xl">
          <div className="mb-2 flex items-center justify-between">
            <p className="font-bold">Offers</p>
            <button type="button" aria-label="Close offers" onClick={() => s.setOffersOpen(false)}>
              <CloseIcon />
            </button>
          </div>
          {[
            ["FLASH30", "30% off the flash sale"],
            ["WELCOME10", "10% off your first order"],
          ].map(([code, label]) => (
            <button key={code} type="button" onClick={() => copy(code)} className="mt-2 flex w-full items-center justify-between rounded-2xl bg-[#f4f691] px-3 py-3 text-left">
              <span>
                <span className="block text-xs">{label}</span>
                <span className="font-bold">{code}</span>
              </span>
              <span className="text-xs font-bold">{copied === code ? "Copied" : "Copy"}</span>
            </button>
          ))}
        </div>
      ) : null}
      <button type="button" onClick={() => s.setOffersOpen(!s.offersOpen)} className="rounded-full bg-black px-4 py-2 text-sm font-bold text-white">
        Offers
      </button>

function FlashSale() {
  const [left, setLeft] = useState(calc());
  const [copied, setCopied] = useState(false);
  const store = useStore();
  useEffect(() => {
    const t = window.setInterval(() => setLeft(calc()), 1000);
    return () => window.clearInterval(t);
  }, []);
  const parts = [
    ["Days", Math.floor(left / 86400)],
    ["Hours", Math.floor((left % 86400) / 3600)],
    ["Mins", Math.floor((left % 3600) / 60)],
    ["Secs", left % 60],
  ];
  return (
    <section className="mx-auto max-w-[1680px] px-4 sm:px-6 lg:px-10">
      <div className="flex flex-col items-center justify-between gap-6 rounded-3xl bg-[#f4f691] px-6 py-8 lg:flex-row lg:px-10">
        <div>
          <p className="text-lg font-bold">Flash Sale now on!</p>
          <p className="text-sm text-[#333]">Save on modern table office, best sellers + more</p>
        </div>
        <div className="flex gap-3">
          {parts.map(([label, n]) => (
            <div key={label as string} className="grid h-[72px] w-[68px] place-items-center rounded-2xl bg-white/70 text-center">
              <span className="text-2xl font-bold leading-none">{String(n).padStart(2, "0")}</span>
              <span className="text-[11px] font-semibold uppercase">{label}</span>
            </div>
          ))}
        </div>
        <button
          type="button"
          className="h-12 rounded-full bg-black px-6 text-sm font-bold text-white"
          onClick={() => {
            navigator.clipboard?.writeText("FLASH30").catch(() => {});
            store.applyCode("FLASH30");
            setCopied(true);
          }}
        >
          {copied ? "Copied" : "Use Code: FLASH30"}
        </button>
      </div>
    </section>
  );
}

function calc() {
  const end = new Date();
  end.setMonth(end.getMonth() + 1, 1);
  end.setHours(0, 0, 0, 0);
  return Math.max(0, Math.floor((end.getTime() - Date.now()) / 1000));
}

import { useEffect, useMemo, useState, type ReactNode } from "react";
import { Link, useParams, useSearchParams } from "react-router-dom";

  collectionProducts,

export function CollectionPage() {
  const { handle = "all" } = useParams();
  const [params, setParams] = useSearchParams();
  const col = getCollection(handle) ?? getCollection("all")!;
  const layout = params.get("layout") || "banner";
  const [sort, setSort] = useState("featured");
  const [grid, setGrid] = useState<"grid" | "list">("grid");
  const [cols, setCols] = useState(4);
  const [maxPrice, setMaxPrice] = useState(4000);
  const [inStock, setInStock] = useState(false);
  const [types, setTypes] = useState<string[]>([]);
  const [colors, setColors] = useState<string[]>([]);
  const [openFilters, setOpenFilters] = useState(false);

  const base = collectionProducts(col.handle);
  const allTypes = Array.from(new Set(base.map((p) => p.type))).sort();
  const allColors = Array.from(
    new Set(base.flatMap((p) => colorOption(p)?.values ?? [])),
  ).sort();

  const list = useMemo(() => {
    let items = base.filter((p) => p.variants[0].price <= maxPrice);
    if (inStock) items = items.filter((p) => p.stock > 0 && !p.tags.includes("Coming Soon"));
    if (types.length) items = items.filter((p) => types.includes(p.type));
    if (colors.length) items = items.filter((p) => colorOption(p)?.values.some((c) => colors.includes(c)));
    const arr = [...items];
    if (sort === "az") arr.sort((a, b) => a.title.localeCompare(b.title));
    if (sort === "za") arr.sort((a, b) => b.title.localeCompare(a.title));
    if (sort === "low") arr.sort((a, b) => a.variants[0].price - b.variants[0].price);
    if (sort === "high") arr.sort((a, b) => b.variants[0].price - a.variants[0].price);
    if (sort === "new") arr.sort((a, b) => Number(b.tags.includes("New")) - Number(a.tags.includes("New")));
    return arr;
  }, [base, maxPrice, inStock, types, colors, sort]);

  const toggle = (arr: string[], v: string, set: (n: string[]) => void) =>
    set(arr.includes(v) ? arr.filter((x) => x !== v) : [...arr, v]);

  const banner = (
    <div
      className={cn(
        "relative overflow-hidden rounded-3xl",
        layout === "none" && "hidden",
        layout === "bg" && "bg-[#1d349a] text-white",
        layout === "cards" && "bg-[#f6f6f6]",
      )}
    >
      {layout !== "bg" && layout !== "cards" ? (
        <img src={file(col.image, 1800)} alt="" className="h-56 w-full object-cover sm:h-72" referrerPolicy="no-referrer" />
      ) : (
        <div className="h-48" />
      )}
      <div className={cn("absolute inset-0 flex flex-col justify-end p-8", layout === "bg" || layout === "cards" ? "relative" : "bg-gradient-to-t from-black/55 to-transparent text-white")}>
        <p className="text-sm font-bold opacity-80">{base.length} products</p>
        <h1 className="text-4xl font-bold">{col.title}</h1>
        <p className="mt-1 max-w-xl text-sm opacity-90">{col.blurb}</p>
      </div>
    </div>
  );

  const filters = (
    <div className="space-y-6">
      <div>
        <p className="font-bold">Availability</p>
        <label className="mt-2 flex items-center gap-2 text-sm">
          <input type="checkbox" checked={inStock} onChange={() => setInStock((v) => !v)} /> In stock ({base.filter((p) => p.stock > 0).length})
        </label>
      </div>
      <div>
        <p className="font-bold">Price</p>
        <input type="range" min={10} max={4000} value={maxPrice} onChange={(e) => setMaxPrice(Number(e.target.value))} className="mt-3 w-full accent-black" />
        <p className="text-sm text-[#666]">Up to ${maxPrice.toLocaleString()}</p>
      </div>
      <div>
        <p className="font-bold">Color</p>
        <div className="mt-2 flex flex-wrap gap-2">
          {allColors.map((c) => (
            <button key={c} type="button" onClick={() => toggle(colors, c, setColors)} className={cn("rounded-full", colors.includes(c) && "ring-2 ring-black ring-offset-2")}>
              <Swatch name={c} />
            </button>
          ))}
        </div>
      </div>
      <div>
        <p className="font-bold">Product type</p>
        <ul className="mt-2 space-y-1 text-sm">
          {allTypes.map((t) => (
            <li key={t}>
              <label className="flex items-center gap-2">
                <input type="checkbox" checked={types.includes(t)} onChange={() => toggle(types, t, setTypes)} />
                {t}
                <span className="text-[#999]">({base.filter((p) => p.type === t).length})</span>
              </label>
            </li>
          ))}
        </ul>
      </div>
      <button
        type="button"
        className="text-sm font-bold underline"
        onClick={() => {
          setTypes([]);
          setColors([]);
          setInStock(false);
          setMaxPrice(4000);
        }}
      >
        Clear filters
      </button>
    </div>
  );

  return (
    <div className="mx-auto max-w-[1680px] px-4 py-8 sm:px-6 lg:px-10">
      <Crumbs items={[{ label: col.title }]} />
      <div className={cn("mb-8", layout === "right" && "lg:ml-auto lg:max-w-3xl", layout === "left" && "lg:max-w-3xl")}>{banner}</div>
      <div className="grid gap-8 lg:grid-cols-[240px_1fr]">
        <aside className="hidden lg:block">{filters}</aside>
        <div>
          <div className="mb-5 flex flex-wrap items-center gap-3">
            <button type="button" className="h-10 rounded-full bg-[#ededed] px-4 text-sm font-bold lg:hidden" onClick={() => setOpenFilters(true)}>
              Filter
            </button>
            <p className="text-sm text-[#666]">{list.length} products</p>
            <div className="ml-auto flex flex-wrap items-center gap-2">
              <select value={sort} onChange={(e) => setSort(e.target.value)} className="h-10 rounded-full bg-[#ededed] px-3 text-sm font-bold">
                <option value="featured">Featured</option>
                <option value="new">New</option>
                <option value="az">Alphabetically, A–Z</option>
                <option value="za">Alphabetically, Z–A</option>
                <option value="low">Price, low to high</option>
                <option value="high">Price, high to low</option>
              </select>
              <button type="button" className={cn("h-10 rounded-full px-3 text-sm font-bold", grid === "grid" ? "bg-black text-white" : "bg-[#ededed]")} onClick={() => setGrid("grid")}>
                Grid
              </button>
              <button type="button" className={cn("h-10 rounded-full px-3 text-sm font-bold", grid === "list" ? "bg-black text-white" : "bg-[#ededed]")} onClick={() => setGrid("list")}>
                List
              </button>
              <button type="button" className="hidden h-10 rounded-full bg-[#ededed] px-3 text-sm font-bold xl:inline" onClick={() => setCols(cols === 4 ? 3 : 4)}>
                {cols} col
              </button>
            </div>
          </div>
          {list.length === 0 ? (
            <p className="py-16 text-center text-[#666]">No pieces match those filters.</p>
          ) : (
            <div className={cn("grid gap-5", grid === "list" ? "grid-cols-1" : cols === 3 ? "grid-cols-2 xl:grid-cols-3" : "grid-cols-2 xl:grid-cols-4")}>
              {list.map((p) => (
                <ProductCard key={p.handle} product={p} layout={grid} swatch={params.get("swatch") === "image" ? "image" : "color"} />
              ))}
            </div>
          )}
        </div>
      </div>
      {openFilters ? (
        <div className="fixed inset-0 z-50 lg:hidden">
          <button className="absolute inset-0 bg-black/40" onClick={() => setOpenFilters(false)} aria-label="Close filters" />
          <div className="absolute inset-y-0 left-0 w-[min(100%,340px)] overflow-auto bg-white p-5">
            <div className="mb-4 flex items-center justify-between">
              <p className="font-bold">Filter</p>
              <button type="button" onClick={() => setOpenFilters(false)}>Close</button>
            </div>
            {filters}
          </div>
        </div>
      ) : null}
      <div className="mt-8 flex flex-wrap gap-2 text-xs">
        {["banner", "left", "right", "bg", "cards", "none"].map((l) => (
          <button key={l} type="button" className="rounded-full bg-[#f3f3f3] px-3 py-1 font-bold capitalize" onClick={() => setParams({ layout: l })}>
            {l} layout
          </button>
        ))}
      </div>
    </div>
  );
}

export function ProductPage() {
  const { handle = "" } = useParams();
  const [params] = useSearchParams();
  const product = getProduct(handle);
  const gallery = params.get("gallery") || "vertical";
  const swatch = params.get("swatch") === "image" ? "image" : "color";
  const store = useStore();
  const colors = product ? colorOption(product) : undefined;
  const other = product?.options.find((o) => o.name !== colors?.name);
  const [color, setColor] = useState(colors?.values[0]);
  const [mat, setMat] = useState(other?.values[0]);
  const [qty, setQty] = useState(1);
  const [shot, setShot] = useState(0);
  const [tab, setTab] = useState("Description");
  const [zip, setZip] = useState("");
  const [eta, setEta] = useState("");
  const [notify, setNotify] = useState("");
  const [noted, setNoted] = useState(false);
  const [showSticky, setShowSticky] = useState(false);

  useEffect(() => {
    if (!product) return;
    setColor(colorOption(product)?.values[0]);
    setMat(product.options.find((o) => o.name !== colorOption(product)?.name)?.values[0]);
    setQty(1);
    setShot(0);
    setTab("Description");
    setNoted(false);
  }, [product]);

  useEffect(() => {
    const onScroll = () => setShowSticky(window.scrollY > 580);
    window.addEventListener("scroll", onScroll, { passive: true });
    return () => window.removeEventListener("scroll", onScroll);
  }, []);

  if (!product) {
    return (
      <div className="mx-auto max-w-3xl px-6 py-24 text-center">
        <h1 className="text-3xl font-bold">Piece not found</h1>
        <Btn to="/collections/all" className="mt-6">
          Back to the collection
        </Btn>
      </div>
    );
  }

  const variant = pickVariant(product, {
    ...(colors && color ? { [colors.name]: color } : {}),
    ...(other && mat ? { [other.name]: mat } : {}),
  });
  const images = product.images.length ? product.images : [variant.image || ""];
  const active = images[Math.min(shot, images.length - 1)];
  const coming = product.tags.includes("Coming Soon");
  const sold = product.stock === 0 || product.handle === "victoria-ottoman" || !variant.available;
  const pct = discountPct(variant.price, variant.compare);
  const more = related(product, 4);

  return (
    <div className="mx-auto max-w-[1680px] px-4 py-8 sm:px-6 lg:px-10">
      <Crumbs items={[{ label: product.type, to: `/collections/${product.type === "Lamps" ? "lighting" : "all"}` }, { label: product.title }]} />
      <div className="grid gap-10 lg:grid-cols-2">
        <Gallery images={images} active={active} shot={shot} setShot={setShot} gallery={gallery} title={product.title} />
        <div className="lg:sticky lg:top-36 lg:self-start">
          <div className="flex items-start justify-between gap-4">
            <div>
              {pct > 0 ? <p className="text-sm font-bold text-[#c4301c]">Save {pct}%</p> : null}
              <h1 className="text-3xl font-bold sm:text-4xl">{product.title}</h1>
              <p className="mt-1 text-sm text-[#666]">
                Vendor: Garace · Type: {product.type} · SKU: {product.sku}
              </p>
            </div>
            <button type="button" aria-label="Save" onClick={() => store.toggleWish(product.handle)} className={store.wished(product.handle) ? "text-[#c4301c]" : ""}>
              <Heart filled={store.wished(product.handle)} />
            </button>
          </div>
          <div className="mt-3 flex items-center gap-3">
            <Stars value={product.rating} />
            <span className="text-sm text-[#666]">{product.rating} · {product.reviews} reviews</span>
          </div>
          <div className="mt-4 text-2xl">
            <Money amount={variant.price} compare={variant.compare} />
          </div>
          <div className="mt-4 flex flex-wrap gap-4 text-sm font-semibold">
            {["Modern", "Eco-certified", "Warranty"].map((t) => (
              <span key={t} className="inline-flex items-center gap-2">
                <span className="grid h-8 w-8 place-items-center rounded-full bg-[#ededed] text-[#1d349a]">✦</span>
                {t}
              </span>
            ))}

function Gallery({
  images,
  active,
  shot,
  setShot,
  gallery,
  title,
}: {
  images: string[];
  active: string;
  shot: number;
  setShot: (n: number) => void;
  gallery: string;
  title: string;
}) {
  if (gallery === "grid") {
    return (
      <div className="grid grid-cols-2 gap-3">
        {images.map((src, i) => (
          <img key={src + i} src={img(src, 900)} alt={title} className="aspect-square w-full rounded-2xl object-cover bg-[#f3f3f3]" referrerPolicy="no-referrer" />
        ))}
      </div>
    );
  }
  if (gallery === "mix") {
    return (
      <div className="grid grid-cols-2 gap-3">
        <img src={img(images[0], 1000)} alt={title} className="col-span-2 aspect-[4/3] w-full rounded-2xl object-cover bg-[#f3f3f3]" referrerPolicy="no-referrer" />
        {images.slice(1).map((src, i) => (
          <img key={src + i} src={img(src, 700)} alt="" className="aspect-square w-full rounded-2xl object-cover bg-[#f3f3f3]" referrerPolicy="no-referrer" />
        ))}
      </div>
    );
  }
  const thumbs = (
    <div className={cn("flex gap-2", gallery === "vertical" ? "flex-col" : "mt-3 flex-row")}>
      {images.map((src, i) => (
        <button key={src + i} type="button" onClick={() => setShot(i)} className={cn("overflow-hidden rounded-xl bg-[#f3f3f3]", gallery === "vertical" ? "h-20 w-20" : "h-16 w-16", i === shot && "ring-2 ring-black")}>
          <img src={img(src, 160)} alt="" className="h-full w-full object-cover" referrerPolicy="no-referrer" />
        </button>
      ))}
    </div>
  );
  return (
    <div className={cn(gallery === "vertical" && "flex gap-3")}>
      {gallery === "vertical" ? thumbs : null}
      <div className="relative min-w-0 flex-1 overflow-hidden rounded-3xl bg-[#f3f3f3]">
        <img src={img(active, 1400)} alt={title} className="aspect-square w-full object-cover" referrerPolicy="no-referrer" />
      </div>
      {gallery !== "vertical" ? thumbs : null}
    </div>
  );
}

export function SearchPage() {
  const [params] = useSearchParams();
  const q = params.get("q") || "";
  const type = params.get("type") || "All Categories";
  const hits = products.filter((p) => {
    const okType = !type || type === "All Categories" || p.type === type;
    const okQ = !q || `${p.title} ${p.type} ${p.tags.join(" ")}`.toLowerCase().includes(q.toLowerCase());
    return okType && okQ;
  });
  return (
    <div className="mx-auto max-w-[1680px] px-4 py-10 sm:px-6 lg:px-10">
      <h1 className="text-3xl font-bold">Search {q ? `“${q}”` : ""}</h1>
      <p className="mt-1 text-sm text-[#666]">{hits.length} results {type !== "All Categories" ? `in ${type}` : ""}</p>
      <div className="mt-8 grid grid-cols-2 gap-4 lg:grid-cols-4">
        {hits.map((p) => (
          <ProductCard key={p.handle} product={p} />
        ))}
      </div>
      {hits.length === 0 ? <p className="py-16 text-[#666]">Nothing matched. Try “chair”, “lamp”, or “table”.</p> : null}
    </div>
  );
}

export function CheckoutPage() {
  const s = useStore();
  const [placed, setPlaced] = useState<string | null>(null);
  useEffect(() => {
    s.setCartOpen(false);
    // close the drawer once on arrival
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);
  const [ship, setShip] = useState(s.subtotal >= 500 ? "express" : "standard");
  const shipping = ship === "express" ? (s.subtotal >= 500 ? 0 : 45) : ship === "pickup" ? 0 : 29;
  const extra = s.subtotal >= 1000 ? 20 : 0;
  const due = Math.max(0, s.total + shipping - extra);
  if (placed) {
    return (

        <Btn to="/" className="mt-6">
          Back to Garace
        </Btn>
      </div>
    );
  }
  return (
    <div className="mx-auto grid max-w-5xl gap-10 px-4 py-10 lg:grid-cols-[1.2fr_.8fr] sm:px-6">
      <form
        className="space-y-4"

          <Field name="last" label="Last name" />
        </div>
        <Field name="address" label="Address" />

          <Field name="city" label="City" />
          <Field name="region" label="State" />
          <Field name="zip" label="ZIP" />

        <fieldset className="space-y-2 rounded-2xl border border-[#eee] p-4">
          <legend className="px-1 text-sm font-bold">Shipping</legend>
          {[
            ["standard", "Standard — $29"],
            ["express", s.subtotal >= 500 ? "Express — Free over $500" : "Express — $45"],
            ["pickup", "In-store pickup — Free"],
          ].map(([id, label]) => (

            <Field name="cvc" label="CVC" placeholder="123" />
          </div>
        </fieldset>
        <Btn type="submit" className="w-full" disabled={s.cart.length === 0}>
          {s.cart.length === 0 ? "Cart is empty" : `Pay ${due.toFixed(2)}`}
        </Btn>
      </form>
      <aside className="h-fit rounded-3xl bg-[#f7f7f7] p-5">
        {s.cart.map((l) => {
          const p = getProduct(l.handle);
          if (!p) return null;
          return (
            <div key={l.variantId} className="mb-3 flex items-center gap-3">
              <img src={img(p.images[0], 120)} alt="" className="h-16 w-16 rounded-xl object-cover" referrerPolicy="no-referrer" />

            <span>Total</span>

function Field({ name, label, type = "text", placeholder }: { name: string; label: string; type?: string; placeholder?: string }) {

  );
}

export function ProductCard({
  product,
  layout = "grid",
  swatch = "color",
}: {
  product: Product;
  layout?: "grid" | "list";
  swatch?: "color" | "image";
}) {
  const { add, toggleWish, wished, toggleCompare, compare, setQuick } = useStore();
  const nav = useNavigate();
  const [hover, setHover] = useState(false);
  const colors = colorOption(product);
  const [color, setColor] = useState(colors?.values[0]);
  const variant = findVariant(product, color);
  const range = priceRange(product);
  const shot = variant.image || product.images[0];
  const alt = product.images.find((i) => i !== shot) || product.images[1];
  const marks = badges(product);
  const coming = product.tags.includes("Coming Soon") || product.stock === 0;
  const saved = wished(product.handle);
  const compared = compare.includes(product.handle);

  const choose = () => {
    if (coming) {
      nav(`/products/${product.handle}`);
      return;
    }
    if ((product.options[0]?.values.length ?? 0) > 1) setQuick(product.handle);
    else add(product.handle, variant.id, 1);
  };

  return (
    <article
      className={cn("group", layout === "list" && "grid grid-cols-[180px_1fr] gap-5 sm:grid-cols-[240px_1fr]")}
      onMouseEnter={() => setHover(true)}
      onMouseLeave={() => setHover(false)}
    >
      <div className="relative overflow-hidden rounded-2xl bg-[#f3f3f3]">
        <Link to={`/products/${product.handle}`} className="block aspect-square">
          <img
            src={img(shot, 800)}
            alt={product.title}
            referrerPolicy="no-referrer"
            className={cn(
              "h-full w-full object-cover transition duration-500",
              hover && alt ? "opacity-0" : "opacity-100",
              hover && "scale-[1.03]",
            )}
          />
          {alt ? (
            <img
              src={img(alt, 800)}
              alt=""
              referrerPolicy="no-referrer"
              className={cn(
                "absolute inset-0 h-full w-full object-cover transition duration-500",
                hover ? "opacity-100" : "opacity-0",
              )}
            />
          ) : null}
        </Link>
        <div className="absolute left-3 top-3 flex flex-col items-start gap-1.5">
          {marks.map((b) => (
            <span
              key={b.label}
              className="rounded-full px-2.5 py-1 text-[11px] font-bold uppercase tracking-wide text-white"
              style={{ background: b.bg }}
            >
              {b.label}
            </span>
          ))}
        </div>
        <div className="absolute right-3 top-3 flex flex-col gap-2 opacity-0 transition group-hover:opacity-100">
          <button
            type="button"
            aria-label="Quick view"
            onClick={() => setQuick(product.handle)}
            className="grid h-10 w-10 place-items-center rounded-full bg-white shadow-sm"
          >
            <svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="1.7">
              <path d="M2 12s4-6 10-6 10 6 10 6-4 6-10 6S2 12 2 12z" />
              <circle cx="12" cy="12" r="2.5" />

            type="button"
            aria-label="Wishlist"

            <Heart filled={saved} />

              return (
                <Swatch
                  key={c}
                  name={c}
                  active={color === c}
                  mode={swatch}
                  image={v.image || product.images[0]}
                  onClick={() => setColor(c)}
                />
              );
            })}
            {colors.values.length > 5 ? (
              <span className="text-xs text-[#666]">+{colors.values.length - 5}</span>
            ) : null}
          </div>
        ) : null}
        {layout === "list" ? (
          <p className="mt-3 max-w-xl text-sm text-[#666]">{product.desc}</p>
        ) : null}
      </div>
    </article>
  );
}

export function formatMoney(amount: number, code = "USD") {
  const c = CURRENCIES.find((x) => x.code === code) ?? CURRENCIES[0];
  const v = amount * c.rate;
  if (c.code === "JPY" || c.code === "KRW") {
    return `${c.symbol}${Math.round(v).toLocaleString()}`;
  }
  return `${c.symbol}${v.toLocaleString(undefined, {
    minimumFractionDigits: 2,
    maximumFractionDigits: 2,
  })}`;
}

export function priceRange(p: Product) {
  const prices = p.variants.map((v) => v.price);
  const min = Math.min(...prices);
  const max = Math.max(...prices);
  const compare =
    p.variants.find((v) => v.compare && v.compare > v.price)?.compare ?? null;
  return { min, max, compare, from: min !== max };
}

export function isSale(p: Product) {
  return p.variants.some((v) => v.compare && v.compare > v.price);
}

export function discountPct(price: number, compare?: number | null) {
  if (!compare || compare <= price) return 0;
  return Math.round((1 - price / compare) * 100);
}

export type Badge = { label: string; bg: string };

export function badges(p: Product): Badge[] {
  const out: Badge[] = [];
  if (p.tags.includes("Coming Soon")) out.push({ label: "Coming Soon", bg: "#7A34D6" });
  else if (p.stock === 0) out.push({ label: "Sold Out", bg: "#ADADAD" });
  else if (isSale(p)) out.push({ label: "Sale", bg: "#C4301C" });
  if (p.tags.includes("New")) out.push({ label: "New", bg: "#0d8756" });
  if (p.stock > 0 && p.stock <= 3) out.push({ label: "Selling Fast", bg: "#1d349a" });
  else if (p.tags.includes("Best Choice!")) out.push({ label: "Best Choice!", bg: "#1d349a" });
  return out.slice(0, 2);
}

export function colorOption(p: Product) {
  return p.options.find((o) => /color|colour/i.test(o.name));
}

export function materialOption(p: Product) {
  return p.options.find((o) => /material|denomination/i.test(o.name) || o.name !== colorOption(p)?.name);
}

export function findVariant(p: Product, o1?: string | null, o2?: string | null) {
  if (o1 && o2) {
    return p.variants.find((v) => v.option1 === o1 && v.option2 === o2) ?? p.variants[0];
  }
  if (o1) return p.variants.find((v) => v.option1 === o1) ?? p.variants[0];
  if (o2) return p.variants.find((v) => v.option2 === o2 || v.option1 === o2) ?? p.variants[0];
  return p.variants[0];
}

export function pickVariant(p: Product, selected: Record<string, string>) {
  return (
    p.variants.find((v) =>
      p.options.every((opt, idx) => {
        const want = selected[opt.name];
        if (!want) return true;
        const got = idx === 0 ? v.option1 : v.option2;
        return got === want;
      }),
    ) ?? p.variants[0]
  );
}

}

export function ComparePage() {
  const { compare, toggleCompare, clearCompare, currency } = useStore();
  const list = compare.map((h) => getProduct(h)).filter(Boolean);
  if (list.length === 0) {
    return (
      <div className="px-6 py-24 text-center">
        <h1 className="text-3xl font-bold">Compare</h1>
        <p className="mt-2 text-[#666]">Select up to five pieces from any collection.</p>
        <Btn to="/collections/chairs" className="mt-6">Browse chairs</Btn>
      </div>
    );
  }
  const rows: [string, (p: NonNullable<(typeof list)[number]>) => string][] = [
    ["Price", (p) => formatMoney(p.variants[0].price, currency)],
    ["Type", (p) => p.type],
    ["Rating", (p) => String(p.rating)],
    ["Dimensions", (p) => p.dims],
    ["Seat", (p) => p.seat],
    ["Weight", (p) => p.weight],
    ["Origin", (p) => p.origin],
    ["Stock", (p) => (p.stock ? `${p.stock} in studio` : "Waitlist")],
  ];
  return (
    <div className="mx-auto max-w-[1680px] overflow-auto px-6 py-10">
      <div className="mb-4 flex items-center justify-between">
        <h1 className="text-3xl font-bold">Compare</h1>
        <button type="button" className="text-sm font-bold underline" onClick={clearCompare}>Clear all</button>
      </div>
      <table className="w-full min-w-[720px] text-left text-sm">
        <thead>
          <tr>
            <th />
            {list.map((p) => (
              <th key={p!.handle} className="p-3 align-top">
                <img src={img(p!.images[0], 300)} alt="" className="mb-2 h-36 w-full rounded-2xl object-cover" referrerPolicy="no-referrer" />
                <Link to={`/products/${p!.handle}`} className="font-bold hover:underline">{p!.title}</Link>
                <button type="button" className="mt-1 block text-xs underline" onClick={() => toggleCompare(p!.handle)}>Remove</button>
              </th>
            ))}
          </tr>
        </thead>
        <tbody>
          {rows.map(([label, fn]) => (
            <tr key={label} className="border-t">
              <th className="py-3 pr-4 font-bold">{label}</th>
              {list.map((p) => (
                <td key={p!.handle} className="p-3">{fn(p!)}</td>
              ))}
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  );
}

                    className="h-10 w-20 rounded-xl bg-[#ededed] px-2"

      <Btn
        className="mt-6"

          rows.forEach((p) => {

            if (n > 0) add(p.handle, p.variants[0].id, n);
          });
        }}
      >
        Add selected to cart
      </Btn>
    </div>
  );
}
